וַאַתַא לֶמַדַיַנַתַה וַמַלַפ הַוַא לֶהַוַנ בֶכַנַוַשֶתַהַוַנ אַיַכַנַא דֶנַתַהַרַוַנ וַנַאַמַרַוַנ אַיַמַכַא לֶה לֶהַנַא חַכַמַתַא הַדַא וַחַיַלַא
לֶא הַוַא הַנַא בֶרַה דֶנַגַרַא לֶא אַמַה מֶתַקַרַיַא מֶרַיַמ וַאַחַוַהַי יַעַקַוַב וַיַוַסַא וַשֶמַעַוַנ וַיַהַוַדַא
וַאַחַוַתַה כֶלַהַיַנ לֶא הַא לֶוַתַנ אַנַיַנ אַיַמַכַא לֶה הַכַיַל לֶהַנַא הַלַיַנ כֶלַהַיַנ
וַמַתַכַשֶלַיַנ הַוִו בֶה הַו דֶיַנ יַשֶוַע אַמַר לֶהַוַנ לֶיַת נַבַיַא דֶצַעַיַר אַלַא בֶמַדַיַנַתַה וַבִבַיַתַה
וַלַא עַבַד תַמַנ חַיַלַא סֶגַיַאַא מֶטַל לֶא הַיַמַנַוַתַהַוַנ
