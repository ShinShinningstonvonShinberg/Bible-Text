אֶנ גֶּיר מֶלתָֿא דֶּאתֿמַללַתֿ בּיַדֿ מַלַאכֶֿא אֶשתַּררַתֿ וכֻֿל דּשַמעָה וַעבַֿר עלֶיה קַבֶּל פֻּורעָנָא בּכִֿאנֻותָֿא
אַיכַּנָא חנַנ נֶערֻוק אֶנ נַבֿסֶא עַל אַילֶינ דּהֶנֻונ אֶנֻונ חַיַינ הָנֻונ דּשַרִיו מֶנ מָרַנ למֶתֿמַלָלֻו ומֶנ אַילֶינ דּמֶנֶה שמַעו בַּנ אֶשתַּרַרו
כַּדֿ סָהֶדֿ עלַיהֻונ אַלָהָא בָּאתֿוָתָֿא וַבֿתֶֿדֿמרָתָֿא וַבֿחַילֶא משַחלפֶֿא וַבֿפֻֿולָגֶֿא דּרֻוחָא דּקֻודֿשָא דֶּאתִֿיהֶבֿו אַיכֿ צֶבֿיָנֶה
לָא הוָא גֶּיר למַלַאכֶֿא שַעבֶּדֿ עָלמָא דַּעתִֿידֿ דַּעלַוהי ממַללִינַנ

לָא הוָא גֶּיר עַל מַלַאכֶֿא מַשלַט הוָא מַותָּא אֶלָא עַל זַרעֶה דַּאבֿרָהָמ מַשלַט הוָא
