רִישָא דֶּינ דּכֻֿלהֶינ אִיתֿ לַנ רַבּ כֻּומרֶא אַינָא דִּיתֶֿבֿ מֶנ יַמִינָא דּכֻֿורסיָא דּרַבֻּותָֿא בַּשמַיָא
וַהוָא משַמשָנָא דּבֶֿיתֿ קֻודֿשָא וַדֿמַשכּנָא דַּשרָרָא הַו דַּקבַֿע אַלָהָא ולָא בַּרנָשָא
כֻּל רַבּ כֻּומרֶא גֶּיר דּקָאֶמ דַּנקַרֶבֿ קֻורבָּנֶא ודֶֿבֿחֶא מֶטֻל הָנָא זָדֿקָא הוָתֿ אָפֿ להָנָא דּנֶהוֶא אִיתֿ לֶה מֶדֶּמ דַּנקַרֶבֿ
וֶאלֻו בַּארעָא הוָא אָפֿלָא כֻּומרָא הָוֶא הוָא מֶטֻל דִּאיתַֿיהֻונ הוַו כֻּומרֶא דַּמקַרבִֿינ הוַו קֻורבָּנֶא אַיכֿ דַּבֿנָמֻוסָא
הָנֻונ דַּמשַמשִינ לַדֿמֻותָֿא וַלטֶלָניָתָֿא דּהָלֶינ דּבַֿשמַיָא אַיכֿ דֶּאתֶֿאמַר למֻושֶא כַּדֿ עָבֶֿדֿ הוָא מַשכּנָא דַּחזִי וַעבֶֿדֿ כֻּלמֶדֶּמ בַּדֿמֻותָֿא הָי דֶּאתֿחַזיַתֿ לָכֿ בּטֻורָא
הָשָא דֶּינ תֶּשמֶשתָּא דַּמיַתּרָא מֶנ הָי קַבֶּל יֶשֻוע משִיחָא אַכֿמָא דַּמיַתּרָא אָפֿ דִּיַתִֿיקִא הָי דַּעבִֿידֿ בָּה מֶצעָיָא וַבֿשֻוודָּיֶא דַּמיַתּרִינ מֶנ דּהָי אֶתֿיַהבַּתֿ
אֶלֻו גֶּיר אִיתֶֿיה הוָתֿ קַדֿמָיתָּא דּלָא רֶשיָנ לַיתּ הוָא אַתֿרָא להָדֶֿא דּתַֿרתֶּינ
רָשֶא להֻונ גֶּיר וֶאמַר דּהָא יַומָתָֿא אָתֶֿינ אָמַר מָריָא וֶאגֿמֻור עַל בַּיתָּא דּבֶֿיתֿ אִיסרָיֶל ועַל בַּיתָּא דּבֶֿיתֿ יִהֻודָֿא דִּיַתִֿקִא חדַֿתָֿא
לָא אַיכֿ הָי דִּיַתִֿקִא דּיֶהבֶּתֿ לַאבָֿהַיהֻונ בּיַומָא דֶּאחדֶּתֿ בִּאידֿהֻונ וַאפּקֶתֿ אֶנֻונ מֶנ אַרעָא דּמֶצרֶינ מֶטֻל דּהֶנֻונ לָא קַוִיו בּדִֿיַתִֿקִא דִּילי אָפֿ אֶנָא בּסִיתֿ בּהֻונ אֶמַר מָריָא
הָדֶֿא דֶּינ דִּיַתִֿקִא דֶּאתֶּל לבַֿיתָּא דּבֶֿיתֿ אִיסרָיֶל בָּתַֿר יַומָתָֿא הָנֻונ אָמַר מָריָא אֶתּלִיוהי לנָמֻוסי בּמַדּעַיהֻונ ועַל לֶבַּוָתֿהֻונ אֶכֿתּבִֿיוהי וֶאהוֶא להֻונ אֶנָא אַלָהָא והֶנֻונ נֶהוֻונ לִי עַמָא
ולָא נַלֶפֿ אנָש לבַֿר מדִֿינתֶּה אָפֿלָא לַאחֻוהי ונִאמַר דּדַֿע למָריָא מֶטֻל דּכֻֿלהֻונ נֶדּעֻונָני מֶנ זעֻורהֻונ וַעדַֿמָא לקַשִישהֻונ
וֶאחַסֶא אֶנֻונ מֶנ עַולהֻונ וַחטָהַיהֻונ תֻּובֿ לָא אֶתּדּכַֿר להֻונ
בּהָי דֶּאמַר חדַֿתָֿא לקַדֿמָיתָּא אַעתּקָה וַאינָא דַּעתֶֿק וַסֶאבֿ קַרִיבֿ הֻו לַחבָֿלָא
