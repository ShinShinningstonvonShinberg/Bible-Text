וַבֿהָלֶינ יַומָתָֿא אחרָיֶא מַלֶל עַמַנ בַּבֿרֶה דּלֶה סָמ יָרתָּא דּכֻֿלמֶדֶּמ ובֶֿה עבַֿדֿ לעָלמֶא
דּהֻויֻו צֶמחָא דּשֻובֿחֶה וצַלמָא דִּאיתֻֿותֶֿה וַאחִידֿ כֻּל בּחַילָא דּמֶלתֶֿה והֻו בַּקנֻומֶה עבַֿדֿ דֻּוכָּיָא דַּחטָהַינ וִיתֶֿבֿ עַל יַמִינָא דּרַבֻּותָֿא בַּמרַומֶא
והָנָא כֻּלֶה יִרֶבֿ מֶנ מַלַאכֶֿא אַיכֿ מָא דַּמיַתַּר שמָא דִּירֶתֿ מֶנ דִּילהֻונ
לַאינָא גֶּיר מֶנ מַלַאכֶֿא מֶמתֻֿומ אֶמַר אַלָהָא דּבֶֿרי אַנתּ אֶנָא יַומָנָא יִלֶדּתָּכֿ ותֻֿובֿ דֶּאנָא אֶהוֶא לֶה לַאבָֿא והֻו נֶהוֶא לִי לַבֿרָא
תֻּובֿ דֶּינ אֶמַתֿי דּמַעֶל בֻּוכֿרָא לעָלמָא אֶמַר דּלֶה נֶסגּדֻֿונ כֻּלהֻונ מַלַאכֶֿא דַּאלָהָא
עַל מַלַאכֶֿא דֶּינ הָכַֿנָא אֶמַר דַּעבַֿדֿ מַלַאכַֿוהי רֻוח וַמשַמשָנַוהי נֻור יָקדָּא
עַל בּרָא דֶּינ אֶמַר דּכֻֿורסיָכֿ דִּילָכֿ אַלָהָא לעָלַמ עָלמִינ שַבֿטָא פּשִיטָא שַבֿטָא דּמַלכֻּותָֿכֿ
רחֶמתּ כִּאנֻותָֿא וַסנַיתּ עַולָא מֶטֻל הָנָא מַשחָכֿ אַלָהָא אַלָהָכֿ מֶשחָא דּחַדֻֿותָֿא יַתִּיר מֶנ חַבֿרַיכּ
ותֻֿובֿ אַנתּ מֶנ בּרִישִיתֿ סָמתּ שֶתֶֿאסֶיה דַּארעָא וַשמַיָא עבָֿדֿ אִידַֿיכּ אֶנֻונ
הֶנֻונ נֶעבּרֻונ וַאנתּ קַיָמ אַנתּ וכֻֿלהֻונ אַיכֿ נַחתָּא נֶבֿלֻונ
וַאיכֿ תַּכֿסִיתָֿא תֶּעֻופֿ אֶנֻונ הֶנֻונ נֶתֿחַלפֻֿונ וַאנתּ אַיכֿ דִּאיתַֿיכּ אַנתּ וַשנַיכּ לָא נֶגֿמרָנ
למַנ דֶּינ מֶנ מַלַאכֶֿא מֶמתֻֿומ אֶמַר דּתֶֿבֿ מֶנ יַמִיני עדַֿמָא דֶּאסִימ בּעֶלדּבָֿבַֿיכּ כֻּובֿשָא תּחֶיתֿ רֶגֿלַיכּ
לָא הָא כֻּלהֻונ אִיתַֿיהֻונ רֻוחֶא דּתֶֿשמֶשתָּא דּמֶשתַּדֿרִינ בּתֶֿשמֶשתָּא מֶטֻל אַילֶינ דַּעתִֿידִֿינ למִארַתֿ חַיֶא
