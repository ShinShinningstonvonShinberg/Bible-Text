צָבֶֿא אנָא דֶּינ דּתֶֿדּעֻונ אַינָא אַגֻֿונָא אִיתֿ לִי חלָפַֿיכֻּונ וַחלָפֿ הָנֻונ דַּבֿלַדִֿיקִיַא וַחלָפֿ שַרכָּא אַילֶינ דּפַֿרצֻופּי לָא חזַו בַּבֿסַר
דּנֶתֿבַּיֻאונ לֶבַּוָתֿהֻונ ונֶתֿקַרבֻֿונ בּחֻובָּא לכֻֿלֶה עֻותֿרָא דַּפּיָסָא וַלסֻוכָּלָא דִּידַֿעתֶֿה דּארָזָא דַּאלָהָא אַבָֿא ודַֿמשִיחָא
הַו דּבֶֿה כַּסיָנ כֻּלהֶינ סִימָתָֿא דּחֶכֿמתָֿא ודִֿידַֿעתָֿא
הָדֶֿא דֶּינ אָמַר אנָא דּלָא אנָש נֶהוֶא מַטעֶא לכֻֿונ בַּפֿיָסָא דּמֶלֶא
אָפֶֿנ בַּבֿסַר גֶּיר פַּרִיק אנָא מֶנכֻֿונ אֶלָא בּרֻוח עַמכֻֿונ אנָא וחָדֶֿא אנָא דּחָזֶא אנָא מטַכּסֻותֿכֻֿונ ושַרִירֻותָֿא דּהַימָנֻותֿכֻֿונ דּבַֿמשִיחָא
אַיכַּנָא הָכִֿיל דּקַבֶּלתֻּונ ליֶשֻוע משִיחָא מָרַנ בֶּה הַלֶכֿו
כַּדֿ משַררִינ עֶקָרַיכֻּונ ומֶתֿבּנֶינ אנתֻּונ בֶּה ומֶתֿקַימִינ אנתֻּונ בּהַימָנֻותָֿא הָי דִּילֶפֿתֻּונ דּבָֿה תֶּתֿיַתּרֻונ בּתַֿודִּיתָֿא
אֶזדַּהרו דַּלמָא אנָש נחַלֶצכֻֿונ בּפִֿילָסָפֻֿותָֿא וַבֿטָעיֻותָֿא סרִיקתָּא אַיכֿ יֻולפָּנֶא דַּבֿנַינָשָא וַאיכֿ אֶסטֻוכּסַוהי דּעָלמָא ולָא אַיכֿ משִיחָא
דּבֶֿה עָמַר כֻּלֶה מֻולָיָא דַּאלָהֻותָֿא גֻּושמָנָאיִתֿ
ובֶֿה אָפֿ אַנתֻּונ מֶשתַּמלֶיתֻּונ דּהֻו אִיתַֿוהי רִשָא דּכֻֿלהֶינ אַרכַֿוס ושֻולטָנֶא
ובֶֿה אֶתֿגּזַרתֻּונ גּזֻורתָּא דּלָא בִּאידִֿינ בַּשלָח בֶּסרָא דַּחטָהֶא בַּגֿזֻורתֶּה דַּמשִיחָא
וֶאתֿקבַֿרתֻּונ עַמֶה בּמַעמֻודִֿיתָֿא ובֶֿה קָמתֻּונ עַמֶה דּהַימֶנתֻּונ בּחַילָא דַּאלָהָא דַּאקִימֶה מֶנ בֶּיתֿ מִיתֶֿא
וַלכֻֿונ דּמִיתִֿינ הוַיתֻּונ בַּחטָהַיכֻּונ וַבֿעֻורלֻותֿ בֶּסרכֻֿונ אַחִיכֻֿונ עַמֶה וַשבַֿק לַנ כֻּלהֻונ חטָהַינ
וַעטָא בּפֻֿוקדָּנַוהי שטָר חַובַּינ הַו דִּאיתַֿוהי הוָא סַקֻובֿלַנ ושַקלֶה מֶנ מצַעתָֿא וקַבֿעֶה בַּזקִיפֶֿה
ובַֿשלָח פַּגֿרֶה פַּרסִי לַארכַֿוס וַלשַלִיטָנֶא וַאבֿהֶתֿ אֶנֻונ גַּליָאיִתֿ בַּקנֻומֶה
לָא הָכִֿיל אנָש נדַֿוֶדֿכֻֿונ בּמֶאכֿלָא וַבֿמַשתּיָא אַו בּפֻֿולָגֶֿא דּעִאדֶֿא וַדֿרִיש יַרחֶא וַדֿשַבֶּא
הָלֶינ דִּאיתַֿיהֶינ טֶלָניָתָֿא דַּאילֶינ דַּעתִֿידָֿנ פַּגֿרָא דֶּינ משִיחָא הו
וַלמָא אנָש נֶצבֶּא בּמַכִּיכֻֿותֿ רֶעיָנָא לַמחַיָבֻֿותֿכֻֿונ דּתֶֿשתַּעבּדֻֿונ לפֻֿולחָנָא דּמַלַאכֶֿא בַּדֿסָעֶא עַל מֶדֶּמ דּלָא חזָא וַסרִיקָאיִתֿ מֶתֿחתַֿר בּרֶעיָנָא דּבֶֿסרֶה
ולָא אָחֶדֿ רִשָא דּמֶנֶה כֻּלֶה פַּגֿרָא מֶתֿרַכַּבֿ ומֶתֿקַיַמ בּשֶריָנֶא וַבֿהַדָּמֶא ורָבֶֿא תַּרבִּיתָֿא דַּאלָהָא
אֶנ גֶּיר מִיתּתֻּונ עַמ משִיחָא מֶנ אֶסטֻוכּסַוהי דּעָלמָא למָנָא אַיכֿ הַו דּחַיִינ אנתֻּונ בּעָלמָא מֶתּדִֿינִינ אנתֻּונ
לָא לַמ תֶּקרֻובֿ ולָא תֶּטעַמ ולָא תֶּקַפֿ
דִּאיתַֿיהֶינ הָלֶינ חשַחתָֿא דּמֶתֿחַבּלָא ופֻֿוקָדֶֿא אֶנֶינ ויֻולפָּנֶא דַּבֿנַינָשָא
ומֶתֿחַזיָנ דִּאיתֿ בּהֶינ מֶלתָֿא דּחֶכֿמתָֿא בּפַּרצֻופּ מַכִּיכֻֿותָֿא ודֶֿחלַתֿ אַלָהָא וַדֿלָא חָיסִינ עַל פַּגֿרָא לַו בּמֶדֶּמ דַּמיַקַר אֶלָא בַּאילֶינ דַּחשַחתָֿא אֶנֶינ דּבֶֿסרָא
