וָאפֿ לכֻֿונ דּמִיתִֿינ הוַיתֻּונ בַּחטָהַיכֻּונ וַבֿסַכֿלוָתֿכֻֿונ
בּהָלֶינ דּמֶנ קדִֿימ הַלֶכֿתֻּונ הוַיתֻּונ בּהֶינ אַיכֿ עָלמָיֻותֶֿה דּעָלמָא הָנָא וַאיכֿ צֶבֿיָנֶה דּרִיש שֻולטָנָא דָּאאַר וַדֿרֻוחָא הָדֶֿא דּמֶתֿחַפּטָא בַּבֿנַיָא דּלָא מֶתּטפִּיסָנֻותָֿא
בּהָנֻונ עבָֿדֶֿא דָּאפֿ חנַנ אֶתֿהַפַּכֿנ בּהֻונ מֶנ קדִֿימ בַּרגִֿיגָֿתָֿא דּבֶֿסרַנ ועָבֿדִּינ הוַינ צֶבֿיָנָא דּבֶֿסרַנ וַדֿתַֿרעִיתַֿנ וַבֿנַיָא הוַינ דּרֻוגֿזָא מַליָאיִתֿ אַיכֿ שַרכָּא
אַלָהָא דֶּינ דּעַתִּיר בּרַחמַוהי מֶטֻל חֻובֶּה סַגִּיָאא דַּאחבַֿנ
כַּדֿ מִיתִֿינ הוַינ בַּחטָהַינ אַחיַנ עַמ משִיחָא וַבֿטַיבֻּותֶֿה פַּרקַנ
וַאקִימַנ עַמֶה וַאותּבַֿנ עַמֶה בַּשמַיָא בּיֶשֻוע משִיחָא
דַּנחַוֶא לעָלמֶא דָּאתֶֿינ רַבֻּותָֿא דּעֻותֿרָא דּטַיבֻּותֶֿה ובַֿסִימֻותֶֿה דַּהוָתֿ עלַינ בּיֶשֻוע משִיחָא
בּטַיבֻּותֶֿה הֻו גֶּיר אֶתֿפּרֶקנ בּהַימָנֻותָֿא והָדֶֿא לָא הוָתֿ מֶנכֻֿונ אֶלָא מַוהַבֿתָּא הי דַּאלָהָא
לָא מֶנ עבָֿדֶֿא דּלָא אנָש נֶשתַּבֿהַר
בּרִיתָֿא חנַנ גֶּיר דִּילֶה דֶּאתֿבּרִינ בּיֶשֻוע משִיחָא לַעבָֿדֶֿא טָבֶֿא הָנֻונ דּמֶנ קדִֿימ טַיֶבֿ אַלָהָא דַּבֿהֻונ נהַלֶכֿ
מֶטֻל הָדֶֿא הוַיתֻּונ עָהדִּינ דַּאנתֻּונ עַממֶא מֶנ קדִֿימ דַּבֿסַר הוַיתֻּונ ומֶתֿקרֶינ הוַיתֻּונ עֻורלֻותָֿא מֶנ הָי דּמֶתֿקַריָא גּזֻורתָּא וִאיתֶֿיה עבָֿדֿ אִידַֿיָא בּבֶֿסרָא
וִאיתַֿיכֻּונ הוַיתֻּונ בּהַו זַבֿנָא דּלָא משִיחָא ונֻוכֿרָיִינ הוַיתֻּונ מֶנ דֻּובָּרֶא דִּאיסרָיֶל וַאכֿסנָיֶא הֻו לדִֿיַתִֿקִא דּמֻולכָּנָא וַדֿלָא סבַֿר הוַיתֻּונ וַדֿלָא אַלָה בּעָלמָא
הָשָא דֶּינ בּיֶשֻוע משִיחָא אַנתֻּונ דּמֶנ קדִֿימ רַחִיקִינ הוַיתֻּונ קַרִיבֶֿא הוַיתֻּונ בַּדֿמֶה דַּמשִיחָא
הֻויֻו גֶּיר שַינַנ הַו דַּעבַֿדֿ תַּרתַּיהֶינ חדָֿא וַשרָא סיָגָֿא דּקָאֶמ הוָא בַּמצַעתָֿא
וַבֿעֶלדּבָֿבֻֿותָֿא בּבֶֿסרֶה ונָמֻוסָא דּפֻֿוקָדֶֿא בּפֻֿוקדָּנַוהי בַּטֶל דּלַתֿרַיהֻונ נֶבֿרֶא בַּקנֻומֶה לחַדֿ בַּרנָשָא חַדֿתָֿא וַעבַֿדֿ שַינָא
ורַעִי לַתֿרַיהֻונ בּחַדֿ פּגַֿר עַמ אַלָהָא ובַֿזקִיפֶֿה קטַל בּעֶלדּבָֿבֻֿותָֿא
וֶאתָֿא סַבַּר שלָמָא לכֻֿונ לרַחִיקֶא וַלקַרִיבֶֿא
מֶטֻל דּבֶֿה הֻו הוָא לַנ קֻורָבָֿא לַתֿרֶינ בַּחדָֿא רֻוח לוָתֿ אַבָֿא
מֶכִּיל לָא הוַיתֻּונ אַכֿסנָיֶא ולָא תַּותָּבֶֿא אֶלָא בּנַי מדִֿינתָּא דּקַדִּישֶא וַבֿנַי בַּיתֶּה דַּאלָהָא
וֶאתֿבּנִיתֻּונ עַל שֶתֶֿאסתָּא דַּשלִיחֶא ודַֿנבִֿיֶא והֻו הוָא רִיש קַרנָא דּבֶֿניָנָא יֶשֻוע משִיחָא
ובֶֿה מֶתֿרַכַּבֿ כֻּלֶה בֶּניָנָא ורָבֶֿא להַיכּלָא קַדִּישָא בּמָריָא
כַּדֿ אָפֿ אַנתֻּונ בֶּה מֶתֿבּנֶינ אנתֻּונ למַעמרָא דַּאלָהָא בּרֻוח
