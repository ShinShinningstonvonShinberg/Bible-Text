מֶטֻל הָנָא אֶנָא פַּולָוס אַסִירָא אֶנָא דּיֶשֻוע משִיחָא עַל אַפַּיכֻּונ עַממֶא
אֶנ שמַעתֻּונ מדַֿבּרָנֻותָֿא דּטַיבֻּותֶֿה דַּאלָהָא דֶּאתֿיַהבַּתֿ לִי בּכֻֿונ
דַּבֿגֶֿליָנָא אֶתִֿידַֿע לִי ארָזָא אַיכַּנָא דּכֶֿתֿבֶּתֿ לכֻֿונ בַּזעֻוריָתָֿא
אַיכֿ מָא דּמֶשכּחִיתֻּונ כַּדֿ קָרֶינ אנתֻּונ למֶסתַּכָּלֻו יִדַֿעתֿי דַּבֿארָזֶה דַּמשִיחָא

דּנֶהוֻונ עַממֶא בּנַי יָרתֻּותֶֿה ושַותָּפֶֿא דּפַֿגֿרֶה וַדֿמֻולכָּנָא דֶּאתִֿיהֶבֿ בֶּה בּיַדֿ אֶוַנגֶּלִיָונ
הַו דֶּאנָא הוִיתֿ משַמשָנֶה אַיכֿ מַוהַבֿתָּא דּטַיבֻּותֶֿה דַּאלָהָא דֶּאתֿיַהבַּתֿ לִי מֶנ מַעבּדָֿנֻותָֿא דּחַילֶה

וֶאנַהַר לכֻֿלנָש אַידָֿא הָי מדַֿבּרָנֻותָֿא דּארָזָא הַו דַּכֿסֶא הוָא מֶנ עָלמֶא בַּאלָהָא דּכֻֿל בּרָא
דַּבֿיַדֿ עִדּתָּא תֶּתִֿידַֿע חֶכֿמתֶֿה דַּאלָהָא מַליַתֿ פֻּורשָנֶא לַארכַֿוס וַלשֻולטָנֶא דּבַֿשמַיָא
הָי דּעַתֶּדֿ הוָא מֶנ עָלמֶא ועַבֿדָּה בּיֶשֻוע משִיחָא מָרַנ
הַו דּבֶֿה אִיתֿ לַנ פַּרֶהסִיַא וקַרִיבֻֿותָֿא בּתֻֿוכֿלָנָא דּהַימָנֻותֶֿה
מֶטֻל הָנָא שָאֶלנָא דּלָא תֶּמַאנ לִי בֻּאולצָנַי דּעַל אַפַּיכֻּונ דּהָדֶֿא הִי תֶּשבֻּוחתּכֻֿונ

הַו דּמֶנֶה מֶשתַּמהָא כֻּל אַבָֿהֻותָֿא דּבַֿשמַיָא ובַֿארעָא
דּנֶתֶּל לכֻֿונ אַיכֿ עֻותֿרָא דּתֶֿשבֻּוחתֶּה דַּבֿחַילָא תֶּשתַּררֻונ בּרֻוחֶה דַּבֿבַֿרנָשכֻֿונ דַּלגַֿו

דּתֶֿשכּחֻונ למַדֿרָכֻֿו עַמ כֻּלהֻונ קַדִּישֶא מָנַו רַומָא ועֻומקָא וֻאורכָּא וַפֿתָֿיָא
ותֶֿדּעֻונ רַבֻּותָֿא דִּידַֿעתָֿא דּחֻובֶּה דַּמשִיחָא ותֶֿתֿמַלֻונ בּכֻֿלֶה מֻולָיָא דַּאלָהָא
להַו דֶּינ דַּמצֶא בּחַילָא יַתִּיר מֶנ כֻּל למֶעבַּדֿ לַנ ויַתִּיר מֶנ מָא דּשָאלִינַנ ורָנֶינַנ אַיכֿ חַילֶה דּמֶסתּעַר בַּנ
לֶה תֶּשבֻּוחתָּא בּעִדּתֶּה בּיֶשֻוע משִיחָא בּכֻֿלהֻונ דָּרֶא דּעָלמַי עָלמִינ אַמִינ
