וַחזִיתֿ אחרִנָא מַלַאכָֿא דַּנחֶתֿ מֶנ שמַיָא וַמעַטַפֿ ענָנָא וקֶשתָּא דַּשמַיָא עַל רִשֶה וחֶזוֶה אַיכֿ שֶמשָא ורֶגֿלַוהי אַיכֿ עַמֻודֶֿא דּנֻורָא
וִאיתֿ לֶה בִּאידֶֿה כּתָֿבֻֿונָא פּתִֿיחָא וסָמ רֶגֿלֶה דּיַמִינָא עַל יַמָא דּסֶמָלָא דֶּינ עַל אַרעָא
וַקעָא בּקָלָא רָמָא אַיכֿ אַריָא דּגָֿסַר וכַֿדֿ קעָא מַלֶלו שַבֿעָא רַעמִינ בּקָלַיהֻונ
וכַֿדֿ מַלֶלו שַבֿעָא רַעמִינ מטַיֶבֿ הוִיתֿ למֶכֿתַּבֿ ושֶמעֶתֿ קָלָא מֶנ שמַיָא דּשַבֿעָא דָּאמַר חתֻֿומ הַו מָא דּמַלֶלו שַבֿעָא רַעמִינ ולָא תֶּכֿתּבִֿיוהי
ומַלַאכָֿא הַו דַּחזִיתֿ דּקָאֶמ עַל יַמָא ועַל יַבֿשָא דַּארִימ אִידֶֿה לַשמַיָא
וִימָא בּהַו דּחַי לעָלַמ עָלמִינ הַו דַּבֿרָה לַשמַיָא וַדֿבָֿה ולַארעָא וַדֿבָֿה דּתֻֿובֿ זַבֿנָא לָא נֶהוֶא
אֶלָא בּיַומָתָֿא דּמַלַאכָֿא דּשַבֿעָא מָא דַּעתִֿידֿ למֶזעַק וֶאשתַּלַמ ארָזֶה דַּאלָהָא הַו דּסַבַּר לעַבֿדַּוהי נבִֿיֶא
וקָלָא שֶמעֶתֿ מֶנ שמַיָא תֻּובֿ דַּממַלֶל עַמי וָאמַר זֶל סַבֿ לַכֿתָֿבֻֿונָא דּבִֿאידֶֿה דּמַלַאכָֿא דּקָאֶמ עַל אַרעָא ועַל יַמָא
וֶאזֶלתֿ לוָתֿ מַלַאכָֿא כַּדֿ אָמַר אנָא לֶה למֶתַּל לִי לַכֿתָֿבֻֿונָא וֶאמַר לִי סַבֿ וַאכֻֿולָיהי ונַמַר לָכֿ כַּרסָכֿ אֶלָא בּפֻֿומָכֿ נֶהוֶא אַיכֿ דֶּבֿשָא
ונֶסבֶּתֿ לַכֿתָֿבֻֿונָא מֶנ אִידֶֿה דּמַלַאכָֿא וֶאכַֿלתֶּה וִאיתֿ הוָא בּפֻֿומי אַיכֿ דֶּבֿשָא חַליָא וכַֿדֿ אֶכַֿלתֶּה מֶרַתֿ כַּרסי
וֶאמַר לִי יִהִיבֿ לָכֿ תֻּובֿ זַבֿנָא למֶתֿנַבָּיֻו עַל עַממֶא וֶאמוָתָֿא ולֶשָנֶא ומַלכֶּא סַגִּיֶאא
