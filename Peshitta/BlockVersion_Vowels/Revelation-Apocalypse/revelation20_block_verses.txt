וַחזִיתֿ אחרִנָא מַלַאכָֿא דַּנחֶתֿ מֶנ שמַיָא דִּאיתֿ עלַוהי קלִידָֿא דַּתֿהֻומָא ושִישַלתָּא רַבּתָֿא בִּאידֶֿה
ולַבֿכֶּה לתַֿנִינָא חֶויָא קַדֿמָיָא הַו דִּאיתַֿוהי אָכֶֿלקַרצָא וסָטָנָא וַאסרֶה אָלֶפֿ שנִינ
וַארמיֶה בַּתֿהֻומָא וֶאחַדֿ וַטבַֿע לעֶל מֶנֶה דּלָא תֻּובֿ נַטעֶא לכֻֿלהֻונ עַממֶא בָּתַֿר הָלֶינ יִהִיבֿ למֶשריֶה קַלִיל זַבֿנָא
וַחזִיתֿ מַותּבֶֿא וִיתֶֿבֿו עלַיהֻונ ודִֿינָא אֶתִֿיהֶבֿ להֻונ ונַפֿשָתָֿא הָלֶינ דֶּאתֿפּסֶק מֶטֻל סָהדֻּותָֿא דּיֶשֻוע ומֶטֻל מֶלתָֿא דַּאלָהָא ודַֿאילֶינ דּלָא סגֶֿדֿו לחַיֻותָֿא ולָא לצַלמָה ולָא נסַבֿו רֻושמָא עַל בֶּיתֿ עַינַיהֻונ אַו עַל אִידַֿיהֻונ דַּחיַו וַאמלֶכֿו עַמ משִיחָא אָלֶפֿ שנִינ
והָדֶֿא הי קיָמתָּא קַדֿמָיתָּא
טֻובָֿנָא הו וקַדִּישָא מַנ דִּאיתֿ לֶה מנָתָֿא בַּקיָמתָּא קַדֿמָיתָּא ועַל הָלֶינ לַיתּ שֻולטָנָא למַותָּא תֶּניָנָא אֶלָא נֶהוֻונ כָּהנֶא דַּאלָהָא ודַֿמשִיחָא ונַמלכֻֿונ עַמֶה אָלֶפֿ שנִינ
ומָא דֶּאשתַּלַמ אָלֶפֿ שנִינ נֶשתּרֶא סָטָנָא מֶנ חבֻֿושיֶה
ונֶפֻּוק למַטעָיֻו לכֻֿלהֻונ עַממֶא בַּארבַּע זָויָתָֿה דַּארעָא לגֻֿוגֿ וַלמָגֻֿוגֿ ולַמכַֿנָשֻו אֶנֻונ לַקרָבָֿא אַילֶינ דּמֶניָנהֻונ אַיכֿ חָלָא דּיַמָא
וַסלֶקו עַל פּתָֿיָה דַּארעָא וחַדֿרֻוה לַמדִֿינתָּא דּמַשרִיתָֿא דּקַדִּישֶא ולַמדִֿינתָּא חַבִּיבֿתָּא ונֶחתַּתֿ נֻורָא מֶנ שמַיָא מֶנ אַלָהָא וֶאכֿלַתֿ אֶנֻונ
וָאכֶֿלקַרצָא מַטעיָנהֻונ אֶתֿרמִי בּיַמתָֿא דּנֻורָא וכֶֿבֿרִיתָֿא אַיכָּא דּחַיֻותָֿא וַנבִֿיָא דַּגָּלָא ונֶשתַּנקֻונ אִימָמָא ולִליָא לעָלַמ עָלמִינ
וַחזִיתֿ כֻּורסיָא רַבָּא חֶוָרָא ולַדֿיָתֶֿבֿ לעֶל מֶנֶה הַו דּמֶנ קדָֿמ אַפַּוהי עֶרקַתֿ אַרעָא וַשמַיָא וַאתַֿר לָא אֶשתּכַֿח להֻונ
וַחזִיתֿ למִיתֶֿא רַורבֶֿא וַזעֻורֶא דּקָמו קדָֿמ כֻּורסיָא וסֶפֿרֶא אֶתֿפּתַֿחו וַאחרִנָא סֶפֿרָא אֶתֿפּתַֿח דִּאיתַֿוהי דּדִֿינָא וֶאתּדִֿינו מִיתֶֿא מֶנ אַילֶינ דַּכֿתִֿיבָֿנ בּסֶפֿרָא אַיכֿ עבָֿדַֿיהֻונ
ויַהבֿ יַמָא מִיתֶֿא דּבֶֿה ומַותָּא וַשיֻול יַהבֿו מִיתֶֿא דּצֶאידַֿיהֻונ וֶאתּדִֿינ חַדֿ חַדֿ מֶנהֻונ אַיכֿ עבָֿדַֿיהֻונ
ומַותָּא וַשיֻול אֶתֿרמִיו בּיַמתָֿא דּנֻורָא הָנָא דִּאיתַֿוהי מַותָּא תֶּניָנָא
