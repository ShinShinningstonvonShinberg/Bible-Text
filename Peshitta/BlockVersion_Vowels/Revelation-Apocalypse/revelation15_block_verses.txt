וַחזִיתֿ אחרִתָֿא אָתָֿא בַּשמַיָא רַבּתָֿא וַתֿמִיהתָּא מַלַאכֶֿא דִּאיתֿ עלַיהֻונ מַחוָתָֿא שבַֿע אחרָיָתָֿא דַּבֿהֶינ אֶשתַּמליַתֿ חֶמתֶֿה דַּאלָהָא
וַחזִיתֿ אַיכֿ יַמָא דַּזגֻֿוגִֿיתָֿא דַּפֿתִֿיכָֿא בּנֻורָא וַלדַֿזכַֿו מֶנ חַיֻותָֿא ומֶנ צַלמָה ומֶנ מֶניָנָא דַּשמָה דּקָימִינ לעֶל מֶנ יַמָא דַּזגֻֿוגִֿיתָֿא וִאיתֿ עלַיהֻונ קִיתָֿרַוהי דַּאלָהָא
וַמשַבּחִינ תֶּשבֻּוחתָּא דּמֻושֶא עַבֿדֶּה דַּאלָהָא ותֶֿשבֻּוחתָּא דֶּאמרָא וָאמרִינ רַורבִֿינ ותַֿמִיהִינ עבָֿדַֿיכּ מָריָא אַלָהָא אַחִידֿ כֻּל כִּאנִינ ושַרִירִינ עבָֿדַֿיכּ מַלכָּא דּעָלמֶא
מַנ לָא נֶדֿחַל לָכֿ מָריָא וַנשַבַּח לַשמָכֿ מֶטֻל דַּאנתּ הֻו בַּלחֻודֿ חַסיָא מֶטֻל דּכֻֿלהֻונ עַממֶא נִאתֻֿונ ונֶסגּדֻֿונ קדָֿמַיכּ מֶטֻל דַּתֿרִיצ אַנתּ
ומֶנ בָּתַֿר הָלֶינ חזִיתֿ וֶאתֿפּתַֿח הַיכּלָא דּמַשכּנָא דּסָהדֻּותָֿא בַּשמַיָא
וַנפַֿקו שַבֿעָא מַלַאכִֿינ מֶנ הַיכּלָא הָנֻונ דִּאיתֿ עלַיהֻונ שבַֿע מַחוָנ כַּדֿ לבִֿישִינ כֶּתָּנָא דַּכֿיָא וַנהִירָא וַאסִירִינ עַל חַדֿיַיהֻונ אֶסָרָא דּדַֿהבָֿא
וַחדָֿא מֶנ אַרבַּע חַיוָתָֿא יֶהבַּתֿ לשַבֿעָא מַלַאכִֿינ שבַֿע זָבֻֿורִינ דּמַליָנ חֶמתֶֿה דַּאלָהָא דִּאיתַֿוהי חַיָא לעָלַמ עָלמִינ אַמִינ
וֶאתֿמלִי הַיכּלָא מֶנ תּנָנָא דּתֶֿשבֻּוחתֶּה דַּאלָהָא ומֶנ חַילֶה ולַיתּ דַּמצֶא הוָא למֶעַל להַיכּלָא עדַֿמָא דּנֶשתַּמליָנ שבַֿע מַחוָנ דּשַבֿעָא מַלַאכִֿינ
