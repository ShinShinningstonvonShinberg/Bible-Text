וכַֿדֿ פּתַֿח טַבֿעָא דּשַבֿעָא הוָא שֶתֿקָא בַּשמַיָא אַיכֿ פֶּלגֻּותֿ שָעָא
וַחזִיתֿ לשַבֿעָא מַלַאכִֿינ אַילֶינ דַּקדָֿמ אַלָהָא קָימִינ הוַו דֶּאתִֿיהֶבֿו להֻונ שַבֿעָא שִיפֻֿורִינ
וַאחרִנָא מַלַאכָֿא אֶתָֿא וקָמ עַל מַדֿבּחָא וִאיתֿ לֶה פִּירמָא דּדַֿהבָֿא וֶאתִֿיהֶבֿ לֶה בֶּסמֶא סַגִּיֶאא דּנֶתֶּל בַּצלַוָתָֿא דּכֻֿלהֻונ קַדִּישֶא עַל מַדֿבּחָא דַּקדָֿמ כֻּורסיָא
וַסלֶק עֶטרָא דּבֶֿסמֶא בַּצלַוָתָֿא דּקַדִּישֶא מֶנ יַדֿ מַלַאכָֿא קדָֿמ אַלָהָא
וַנסַבֿ מַלַאכָֿא לפִּירמָא וַמלָיהי מֶנ נֻורָא דּעַל מַדֿבּחָא וַארמִי עַל אַרעָא וַהוַו רַעמֶא וקָלֶא ובַֿרקֶא ונַודָּא
ושַבֿעָא מַלַאכִֿינ דַּעלַיהֻונ שַבֿעָא שִיפֻֿורִינ טַיֶבֿו נַפֿשהֻונ למַזעָקֻו
והַו קַדֿמָיָא אַזעֶק וַהוָא בַּרדָֿא ונֻורָא דַּפֿתִֿיכִֿינ בּמַיָא וֶאתֿרמִיו עַל אַרעָא ותֻֿולתָֿה דַּארעָא יִקֶדֿ ותֻֿולתָֿא דִּאילָנֶא יִקֶדֿ וכֻֿל עֶסבָּא דַּארעָא יִקֶדֿ
ודַֿתֿרֶינ זעַק וַהוָא אַיכֿ טֻורָא רַבָּא דּיָקֶדֿ נפַֿל בּיַמָא וַהוָא תֻּולתֶֿה דּיַמָא דּמָא
ומִיתֿ תֻּולתָֿא דּכֻֿל בֶּריָתָֿא דַּבֿיַמָא דִּאיתֿ בֶּה נַפֿשָא ותֻֿולתָֿא דֶּאלפָֿא אֶתֿחַבַּל
ודַֿתֿלָתָֿא זעַק וַנפַֿל מֶנ שמַיָא כַּוכּבָֿא רַבָּא דּיָקֶדֿ אַיכֿ שַלהֶבִֿיתָֿא וַנפַֿל עַל תֻּולתָֿא דּנַהרַוָתָֿא ועַל עַינָתָֿא דּמַיָא
וַשמֶה דּכַֿוכּבָֿא מֶתֶֿאמַר אַפּסִיתֿנָא וַהוָא תֻּולתֿהֻונ דּמַיָא אַיכֿ אַפּסִנתִֿינ וסֻוגָֿאא דַּבֿנַינָשָא מִיתֿו מֶטֻל דֶּאתֿמַרמַרו מַיָא
ודַֿארבּעָא זעַק וַבֿלַע תֻּולתֶֿה דּשֶמשָא ותֻֿולתֶֿה דּסַהרָא ותֻֿולתָֿא דּכַֿוכּבֶֿא וַחשֶכֿו תֻּולתֿהֻונ ויַומָא לָא חַוִי תֻּולתֶֿה ולִליָא הָכֿוָתֿ
ושֶמעֶתֿ לנֶשרָא חַדֿ דּפָֿרַח בַּשמַיָא דָּאמַר וָי וָי וָי לעָמֻורֶיה דַּארעָא מֶנ קָלָא דּשִיפֻֿורֶא דַּתֿלָתָֿא מַלַאכִֿינ דַּעתִֿידִֿינ למַזעָקֻו
