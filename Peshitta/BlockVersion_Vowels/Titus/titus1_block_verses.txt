פַּולָוס עַבֿדֶּה דַּאלָהָא וַשלִיחֶה דּיֶשֻוע משִיחָא בּהַימָנֻותָֿא דַּגֿבַֿיָא דַּאלָהָא וִידַֿעתָֿא דַּשרָרָא דַּבֿדֶֿחלַתֿ אַלָהָא
עַל סַבֿרָא דּחַיֶא דַּלעָלַמ דַּמלַכֿ אַלָהָא שַרִירָא קדָֿמ זַבֿנַוהי דּעָלמָא
וַגֿלָא מֶלתֶֿה בּזַבֿנָה בּיַדֿ כָּרֻוזֻותַֿנ הָי דֶּאתֿהַימנֶתֿ אֶנָא בּפֻֿוקדָּנָא דַּאלָהָא מַחיָנַנ
לטִטָוס בּרָא שַרִירָא בּהַימָנֻותָֿא דּגַֿוָא טַיבֻּותָֿא וַשלָמָא מֶנ אַלָהָא אַבָֿא ומֶנ מָרַנ יֶשֻוע משִיחָא מַחיָנַנ
מֶטֻלהָנָא הו שבַֿקתָּכֿ הוִיתֿ בַּקרִטִא דַּאילֶינ דּחַסִירָנ תַּתֿקֶנ וַתֿקִימ קַשִישֶא בַּמדִֿינָא מדִֿינָא אַיכַּנָא דּפַֿקֶדּתָּכֿ
אַינָא דַּדֿלָא רֶשיָנ הֻו וַהוָא בַּעלָא דַּחדָֿא אַנתּתָֿא וִאיתֿ לֶה בּנַיָא מהַימנֶא דּלָא מצַחֶינ ולָא אִיתַֿיהֻונ דּלָא שֻועבָּדֿ בָּאסֻוטֻותָֿא
חַיָבֿ הֻו גֶּיר קַשִישָא דּנֶהוֶא דּלָא רֶשיָנ אַיכֿ רַבַּיתָּא דַּאלָהָא ולָא נֶהוֶא מֶתּדַּבַּר בּרֶעיָנ נַפֿשֶה ולָא נֶהוֶא חֶמתָֿנ ולָא נֶהוֶא עָבַֿר עַל חַמרָא ולָא תֶּהוֶא רָהטָא אִידֶֿה למֶמחָא ולָא נֶהוֶא רָחֶמ יֻותֿרָנֶא טַנפֶּא
אֶלָא נֶהוֶא רָחֶמ אַכֿסנָיֶא ונֶהוֶא רָחֶמ טָבָֿתָֿא ונֶהוֶא נכֶֿפֿ ונֶהוֶא כִּאינ ונֶהוֶא חַסיָא וַלבִֿיכֿ נַפֿשֶה מֶנ רגִֿיגָֿתָֿא
וַבֿטִיל לֶה עַל יֻולפָּנָא דּמֶלתָֿא דּהַימָנֻותָֿא דּנֶשכַּח אָפֿ לַמבַֿיָאוֻ בּיֻולפָּנֶה חלִימָא וַלמַכָּסֻו לַאילֶינ דּמֶתֿחרֶינ
אִיתֿ גֶּיר סַגִּיֶאא דּלָא מֶשתַּעבּדִֿינ וַסרִיקָנ מֶלַיהֻונ ומַטעֶינ רֶעיָנֶא דַּבֿנַינָשָא יַתִּירָאיִתֿ אַילֶינ דּמֶנ גּזֻורתָּא אֶנֻונ
הָנֻונ דּוָלֶא לַמסַכָּרֻו פֻּומהֻונ בָּתֶּא גֶּיר סַגִּיֶאא מחַבּלִינ ומַלפִֿינ מֶדֶּמ דּלָא וָלֶא מֶטֻל יֻותֿרָנֶא טַנפֶּא
אֶמַר אנָש מֶנהֻונ נבִֿיָא דִּילהֻונ דַּבֿנַי קרִטִא בּכֻֿלזבַֿנ דַּגָּלִינ אֶנֻונ חַיוָתָֿא בִּישָתָֿא וכַֿרסָתָֿא בַּטִילָתָֿא
ושַרִירָאיִתֿ אִיתֶֿיה סָהדֻּותָֿא הָדֶֿא מֶטֻל הָנָא קַשיָאיִתֿ הוַיתּ מַכֶּס להֻונ דּנֶהוֻונ חלִימִינ בּהַימָנֻותָֿא
ולָא נֶתֿרמֻונ לשֻועיָתָֿא דִּיהֻודָֿיֶא וַלפֻֿוקָדֶֿא דַּבֿנַינָשָא דּסָנֶינ שרָרָא
