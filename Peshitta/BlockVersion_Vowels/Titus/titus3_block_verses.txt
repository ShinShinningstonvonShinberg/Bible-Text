וַהוַיתּ מַעהֶדֿ להֻונ דַּלרִשֶא וַלשַלִיטָנֶא נֶשתַּמעֻונ ונֶשתַּעבּדֻֿונ וַדֿנֶהוֻונ עתִֿידִֿינ לכֻֿל עבָֿדֿ טָבֿ
ועַל אנָש לָא נגַֿדּפֻֿונ ולָא נֶהוֻונ מֶתֿכַּתּשִינ אֶלָא נֶהוֻונ מַכִּיכִֿינ וַבֿכֻֿלמֶדֶּמ נחַוֻונ בַּסִימֻותֿהֻונ לוָתֿ כֻּלהֻונ בּנַינָשָא
אָפֿ חנַנ גֶּיר מֶנ קדִֿימ דּלָא רֶעיָנָא הוַינ וַדֿלָא פּיָסָא וטָעֶינ הוַינ וַמשַעבּדִֿינ הוַינ לַארגִֿיגָֿתָֿא משַחלפָֿתָֿא וַבֿבִֿישֻותָֿא ובַֿחסָמָא מֶתֿהַפּכִֿינ הוַינ וַסנִיאיִנ הוַינ אָפֿ חַדֿ לחַדֿ סָנֶינ הוַינ
כַּדֿ דֶּינ אֶתֿגַּליַתֿ בַּסִימֻותֶֿה וַמרַחמָנֻותֶֿה דַּאלָהָא מַחיָנַנ
לָא בַּעבָֿדֶֿא דּזַדִּיקֻותָֿא דַּעבַֿדֿנ אֶלָא בּרַחמַוהי דִּילֶה אַחיַנ בַּסחָתָֿא דּמַולָדָֿא דּמֶנ דּרִיש וַבֿחֻודָּתָֿא דּרֻוחָא דּקֻודֿשָא
הָי דֶּאשַדֿ עלַינ עַתִּירָאיִתֿ בּיַדֿ יֶשֻוע משִיחָא מַחיָנַנ
דַּבֿטַיבֻּותֶֿה נֶזדַּדַּק ונֶהוֶא יָרתֶּא בּסַבֿרָא בּחַיֶא דַּלעָלַמ
מהַימנָא הי מֶלתָֿא וַבֿהָלֶינ צָבֶֿא אנָא דָּאפֿ אַנתּ הוַיתּ משַרַר להֻונ דּנֶתֿבּטֶל להֻונ למֶפֿלַח עבָֿדֶֿא טָבֶֿא אַילֶינ דּהַימֶנו בַּאלָהָא הָלֶינ אֶנֶינ טָבָֿנ ומַותּרָנ לַבֿנַינָשָא
מֶנ בּעָתָֿא דֶּינ סַכֿלתָֿא ומֶנ תַּשעיָתָֿא דּשַרבָֿתָֿא ומֶנ חֶריָנֶא ומֶנ תַּכֿתֻּושֶא דּסָפֿרֶא אֶשתַּאל יֻותֿרָנ גֶּיר לַיתּ בּהֶינ וַסרִיקָנ אֶנֶינ
מֶנ גַּבֿרָא הֶרֶסִיַוטָא מֶנ בָּתַֿר חדָֿא זבַֿנ ותַֿרתֶּינ דּמַרתֶּא אַנתּ לֶה אֶשתַּאל
וַהוַיתּ יָדַֿע דּמַנ דּהָכַֿנָא הו מעַקַמ הֻו וחָטֶא והֻו חַיֶבֿ נַפֿשֶה
מָא דּשַדּרֶתֿ לוָתָֿכֿ לַארטֶמַא אַו לטֻוכִֿיקָוס נֶתֿבּטֶל לָכֿ דּתִֿאתֶֿא לוָתֿי לנִיקָפָּולִיס תַּמָנ גֶּיר סָמֶתֿ בּרֶעיָני דַּאסתֶּא
עַל זִנַא דֶּינ סָפֿרָא ועַל אַפָּלָו נֶתֿבּטֶל לָכֿ דַּתֿלַוֶא אֶנֻונ שַפִּיר דּמֶדֶּמ לָא נֶחסַר להֻונ
ונִאלפֻּונ אָפֿ אַילֶינ דּדִֿילַנ אֶנֻונ דּנֶהוֻונ פָּלחִינ עבָֿדֶֿא טָבֶֿא בּצֶבֿוָתָֿא דָּאלצָנ דּלָא נֶהוֻונ דּלָא פִּארֶא
שָאלִינ בַּשלָמָכֿ כֻּלהֻונ אַילֶינ דּעַמי אֶנֻונ שַאל בַּשלָמָא דּכֻֿלהֻונ אַילֶינ דּרָחמִינ לַנ בּהַימָנֻותָֿא טַיבֻּותָֿא עַמ כֻּלכֻֿונ אַמִינ
