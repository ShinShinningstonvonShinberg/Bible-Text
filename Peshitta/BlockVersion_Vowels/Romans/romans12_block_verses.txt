בָּעֶנָא הָכִֿיל מֶנכֻֿונ אַחַי בּרַחמַוהי דַּאלָהָא דַּתֿקִימֻונ פַּגֿרַיכֻּונ דֶּבֿחתָֿא חַיתָֿא וקַדִּישתָּא וַמקַבַּלתָּא לַאלָהָא בּתֶֿשמֶשתָּא מלִילתָּא
ולָא תֶּתּדַּמֻונ לעָלמָא הָנָא אֶלָא אֶשתַּחלַפֿו בּחֻודָּתָֿא דּרֶעיָנַיכֻּונ וַהוַיתֻּונ פָּרשִינ אַינָא הֻו צֶבֿיָנָא דַּאלָהָא טָבָֿא וַמקַבּלָא וַגֿמִירָא
אָמַר אנָא דֶּינ בּטַיבֻּותָֿא דֶּאתֿיַהבַּתֿ לִי לכֻֿלכֻֿונ דּלָא תֶּהוֻונ מֶתֿרַעֶינ לבַֿר מֶנ מָא דּוָלֶא דּתֶֿתֿרַעֻונ אֶלָא תֶּהוֻונ מֶתֿרַעֶינ בּנַכֿפֻּותָֿא כֻּלנָש אַיכֿ דַּפֿלַגֿ לֶה אַלָהָא הַימָנֻותָֿא בַּמשֻוחתָֿא
אַיכַּנָא גֶּיר דַּבֿחַדֿ פַּגֿרָא הַדָּמֶא סַגִּיֶאא אִיתֿ לַנ וכֻֿלהֻונ הַדָּמֶא לָא הוָא חַדֿ סֻוערָנָא אִיתֿ להֻונ
הָכַֿנָא אָפֿ חנַנ דּסַגִּיֶאא חנַנ חַדֿ חנַנ פּגַֿר בַּמשִיחָא חַדֿ חַדֿ מֶנַנ דֶּינ הַדָּמֶא חנַנ דַּחדָֿדֶֿא
אֶלָא אִיתֿ לַנ מַוהבָֿתָֿא משַחלפָֿתָֿא אַיכֿ טַיבֻּותָֿא דֶּאתֿיַהבַּתֿ לַנ אִיתֿ דַּנבִֿיֻותָֿא אַיכֿ משֻוחתָֿא דּהַימָנֻותֶֿה
וִאיתֿ דּתֶֿשמֶשתָּא אִיתֿ לֶה בּתֶֿשמֶשתֶּה וִאיתֿ דּמַלפָֿנָא הֻו בּיֻולפָּנֶה
וִאיתֿ דַּמבַֿיָאנָא הֻו בּבֻֿויָאֶה וַדֿיָהֶבֿ בַּפֿשִיטֻותָֿא וַדֿקָאֶמ בּרִשָא בַּחפִֿיטֻותָֿא ודַֿמרַחֶמ בַּפֿצִיחֻותָֿא
ולָא נֶהוֶא נכִֿיל חֻובּכֻֿונ אֶלָא הוַיתֻּונ סָנֶינ לבִֿישָתָֿא ומֶתֿנַקפִֿינ לטָבָֿתָֿא
הוַיתֻּונ רָחמִינ לַאחַיכֻּונ ומַחבִֿינ חַדֿ לחַדֿ הוַיתֻּונ מקַדּמִינ מיַקרִינ חַדֿ לחַדֿ
הוַיתֻּונ חפִֿיטִינ ולָא חבַֿנָנִינ הוַיתֻּונ רָתֿחִינ בּרֻוח הוַיתֻּונ פָּלחִינ למָרכֻֿונ
הוַיתֻּונ חָדֶֿינ בּסַבֿרכֻֿונ הוַיתֻּונ מסַיבּרִינ אֻולצָנַיכֻּונ הוַיתֻּונ אַמִינִינ בַּצלֻותָֿא
הוַיתֻּונ מֶשתַּותּפִֿינ לַסנִיקֻותָֿא דּקַדִּישֶא הוַיתֻּונ רָחמִינ אַכֿסנָיֶא
בַּרֶכֿו לרָדֻֿופַֿיכֻּונ בַּרֶכֿו ולָא תּלֻוטֻונ
חדַֿו עַמ דּחָדֶֿינ וַבֿכַֿו עַמ דּבָֿכֶֿינ
ומֶדֶּמ דּמֶתֿרַעֶינ אנתֻּונ עַל נַפֿשכֻֿונ אָפֿ עַל אַחַיכֻּונ ולָא תֶּתֿרַעֻונ רֶעיָנָא רָמָא אֶלָא קַפֿו לַאילֶינ דּמַכִּיכִֿינ ולָא הוַיתֻּונ חַכִּימִינ בּרֶעיָנ נַפֿשכֻֿונ
ולָא תֶּפֿרעֻונ לאנָש בִּישתָּא חלָפֿ בִּישתָּא אֶלָא נֶתֿבּטֶל לכֻֿונ דּתֶֿעבּדֻֿונ טָבָֿתָֿא קדָֿמ בּנַינָשָא כֻּלהֻונ
וֶאנ מֶשכּחָא אַיכֿ דּמֶנ לוָתֿכֻֿונ עַמ כֻּלבַּרנָש שלָמָא עבֶֿדֿו
ולָא הוַיתֻּונ תָּבֿעִינ נַפֿשכֻֿונ חַבִּיבַֿי אֶלָא הַבֿו אַתֿרָא לרֻוגֿזָא כּתִֿיבֿ הֻו גֶּיר דֶּאנ לָא תֶּעבֶּדֿ דִּינָא לנַפֿשָכֿ אֶנָא אֶעבֶּדֿ דִּינָכֿ אָמַר אַלָהָא
וֶאנ כּפֶֿנ בּעֶלדּבָֿבָֿכֿ אַוכֶּלָיהי וֶאנ צהֶא אַשקָיהי וֶאנ הָלֶינ תֶּעבֶּדֿ לֶה גֻּומרֶא דּנֻורָא תּקַבַּר עַל קַרקַפֿתֶֿה
לָא תֶּזכֶּיכֻֿונ בִּישתָּא אֶלָא זכַֿאֻוה לבִֿישתָּא בּטָבֿתָֿא
