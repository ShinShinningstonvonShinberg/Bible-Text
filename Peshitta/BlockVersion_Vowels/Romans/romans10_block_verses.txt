אַחַי צֶבֿיָנָא דּלֶבּי ובָֿעֻותֿי דַּלוָתֿ אַלָהָא חלָפַֿיהֻונ דּנִחֻונ
מַסהֶדֿ אנָא גֶּיר עלַיהֻונ דַּטנָנָא דַּאלָהָא אִיתֿ בּהֻונ אֶלָא לַו בִּידַֿעתָֿא
כִּאנֻותֶֿה גֶּיר דַּאלָהָא לָא יִדַֿעו אֶלָא בּעַו דּכִֿאנֻותָֿא דּנַפֿשהֻונ נקִימֻונ ומֶטֻל הָנָא לכִֿאנֻותֶֿה דַּאלָהָא לָא אֶשתַּעבַּדֿו
סָכֶֿה גֶּיר דּנָמֻוסָא משִיחָא הֻו לכִֿאנֻותָֿא לכֻֿל דַּמהַימֶנ
מֻושֶא גֶּיר הָכַֿנָא כּתַֿבֿ כִּאנֻותָֿא דַּבֿנָמֻוסָא דּמַנ דּנֶעבֶּדֿ הָלֶינ נִחֶא בּהֶינ
כִּאנֻותָֿא דֶּינ דַּבֿהַימָנֻותָֿא הָכַֿנָא אָמרָא דּלָא תִּאמַר בּלֶבָּכֿ דּמַנֻו סלֶק לַשמַיָא וַאחֶתֿ לַמשִיחָא
ומַנֻו נחֶתֿ לַתֿהֻומָא דַּשיֻול וַאסֶק לַמשִיחָא מֶנ בֶּיתֿ מִיתֶֿא
אֶלָא מָנָא אֶמַר קַרִיבֿ הֻו לָכֿ פֶּתֿגָֿמָא לפֻֿומָכֿ וַללֶבָּכֿ הָדֶֿא הי מֶלתָֿא דּהַימָנֻותָֿא דּמַכֿרזִינַנ
וֶאנ תַּודֶּא בּפֻֿומָכֿ בּמָרַנ יֶשֻוע וַתֿהַימֶנ בּלֶבָּכֿ דַּאלָהָא אַקִימֶה מֶנ בֶּיתֿ מִיתֶֿא תִּחֶא
לֶבָּא גֶּיר דַּמהַימֶנ בֶּה מֶזדַּדַּק ופֻֿומָא דּמַודֶּא בֶּה חָיֶא
אֶמַר גֶּיר כּתָֿבָֿא דּכֻֿל דַּמהַימֶנ בֶּה לָא נֶבֿהַתֿ
וַבֿהָדֶֿא לָא פּרַש לָא לִיהֻודָֿיֶא ולָא לַארמָיֶא חַדֿ הֻו גֶּיר מָריָא דּכֻֿלהֻונ דּעַתִּיר בּכֻֿל דּקָרֶא לֶה
כֻּל גֶּיר דּנֶקרֶא שמֶה דּמָריָא נִחֶא
אַיכַּנָא הָכִֿיל נֶקרֻונ לַאינָא דּלָא הַימֶנו בֶּה אַו אַיכַּנָא נהַימנֻונ להַו דּלָא שַמעֻוהי אַו אַיכַּנָא נֶשמעֻונ דּלָא מַכֿרזָנָא
אַו אַיכַּנָא נַכֿרזֻונ אֶנ לָא נֶשתַּלחֻונ אַיכֿ דַּכֿתִֿיבֿ דּמָא יָאיָנ רֶגֿלַיהֻונ דַּמסַבּרַי שלָמָא ודַֿמסַבּרַי טָבָֿתָֿא
אֶלָא לָא הוָא כֻּלהֻונ אֶשתּמַעו לַסבַֿרתָֿא דֶּאוַנגֶּלִיָונ אֶשַעיָא גֶּיר אֶמַר מָרי מַנֻו הַימֶנ לבַֿרתֿ קָלַנ
מֶכִּיל הַימָנֻותָֿא מֶנ מַשמַע אֶדֿנָא הי ומַשמַע אֶדֿנָא מֶנ מֶלתָֿא דַּאלָהָא
אֶלָא אָמַר אנָא דַּלמָא לָא שמַעו והָא בּכֻֿלָה אַרעָא נֶפֿקַתֿ בַּרתֿ קָלהֻונ וַבֿסַופֶּיה דּתִֿבֶֿיל מֶלַיהֻונ
אֶלָא אָמַר אנָא דַּלמָא לָא יִדַֿע אִיסרָיֶל קַדֿמָיָא מֻושֶא הָכַֿנָא אֶמַר אַטֶנכֻֿונ בּעַמ דּלָא עַמ וַבֿעַמָא דּלָא מֶתּטפִּיס אַרגֶּזכֻֿונ
אֶשַעיָא דֶּינ אַמרַח וֶאמַר דֶּאתֿחזִיתֿ לַאילֶינ דּלָא בּעַאוֻני וֶאשתַּכֿחֶתֿ לַאילֶינ דַּעלַי לָא שַאֶלו
