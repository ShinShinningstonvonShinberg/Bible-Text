חַלַפ מֶלַכַא וַרַוַרַבַנַא דֶעַוַמַרַא שֶלַיַא וַנַיַחַא נַעַמַר בֶכַלַה דֶחַלַת אַלַהַא וַדַכַיַוַתַא
הַדַא גַיַר שֶפַיַרַא וַמַקַבַלַא קַדַמ אַלַהַא מֶחַיַנִנ
הַו דֶצֶבַא דֶכַלַהַוַנ בֶנַיַנַשַא נַחַוַנ וַנַתַפַנַוַנ לֶיַדַעַתַא דֶשֶרַרַא
חַד הַו גַיַר אַלַהַא וַחַד הַו מֶצַעַיַא דֶאַלַהַא וַדַבַנַיַנַשַא בֶרַנַשַא יַשֶוַע מֶשֶיַחַא
הַו דֶיַהַב נַפַשֶה פַוַרַקַנַא חַלַפ כֶל אַנַש סֶהַדַוַתַא דֶאַתִת בֶזֶבַנַה
הַי דֶאַנַא אַתַסֶיַמַת כֶרַוַזֶה וַשֶלַיַחַה קַוַשֶתַא אַמַר אַנַא וַלַא מֶדַגַל אַנַא דֶהַוַיַת מֶלַפַנַא דֶעַמִמַא בֶהַיַמַנַוַתַא דֶשֶרַרַא
צֶבַא אַנַא הַכַיַל דֶהַוִו מֶצֶלַיַנ גַבַרַא בֶכַל דֶוַכַא כֶד מֶרַיַמַיַנ אַיַדַיַהַוַנ דֶכַיַאַיַת דֶלַא רַוַגַזַא וַדַלַא מֶחַשֶבַתַא
הַכַנַא אַפ נַשַא בֶאַסֶכַמַא נַכַפַא דֶלַבַוַשַא בֶתַחַמַצֶתַא וַבַנַכַפַוַתַא נַהַוַא תַצֶבַיַתַהַיַנ לֶא בֶגַדַוַלַא וַבַדַהַבַא אַו בֶמַרַגַנַיַתַא אַו בֶנַחַתַא שֶפַיַרַא
אַלַא בֶעַבַדַא טַבַא אַיַכַנַא דֶיַאַא לֶנַשַא דֶמַשֶתַוַדַיַנ דֶחַלַת אַלַהַא
אַנַתִתַא בֶשֶלַיַא הַוַת יַלַפַא בֶכַל שֶוַעַבַד
לֶאַנַתִתַא גַיַר לֶמַלַפַו לֶא מֶפַס אַנַא וַלַא לֶמִמַרַחַו עַל גַבַרַא אַלַא תַהַוַא בֶשֶלַיַא
אַדַמ גַיַר אַתַגַבַל לֶוַקַדַמ וַהַיַדַיַנ חַוַא
