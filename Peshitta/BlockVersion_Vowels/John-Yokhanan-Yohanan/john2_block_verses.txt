וַליַומָא דַּתֿלָתָֿא הוָתֿ מֶשתֻּותָֿא בּקָטנֶא מדִֿינתָּא דַּגֿלִילָא וֶאמֶה דּיֶשֻוע תַּמָנ הוָתֿ
וָאפֿ הֻו יֶשֻוע ותַֿלמִידַֿוהי אֶתֿקרִיו לָה למֶשתֻּותָֿא
וַחסַר הוָא חַמרָא וָאמרָא לֶה אֶמֶה ליֶשֻוע חַמרָא לַיתּ להֻונ
אָמַר לָה יֶשֻוע מָא לִי ולֶכֿי אַנתּתָֿא לָא עדַֿכִּיל אֶתָֿתֿ שָעַתֿי
אָמרָא אֶמֶה לַמשַמשָנֶא מֶדֶּמ דָּאמַר לכֻֿונ עבֶֿדֿו
אִיתֿ הוַי דֶּינ תַּמָנ אַגָֿנֶא דּכִֿאפָֿא שֶתֿ דּסִימָנ לתַֿדֿכִּיתָֿא דִּיהֻודָֿיֶא דָּאחדָּנ תּרֶינ תּרֶינ רֶבֿעִינ אַו תּלָתָֿא
אָמַר להֻונ יֶשֻוע מלַו אֶנֶינ מַיָא לַאגָֿנֶא וַמלַו אֶנֶינ עדַֿמָא לעֶל
אָמַר להֻונ זלֻועו מֶכִּיל וַאיתַּו לרִיש סמָכָֿא וַאיתִּיו
וכַֿדֿ טעֶמ הַו רִיש סמָכָֿא מַיָא הָנֻונ דַּהוַו חַמרָא ולָא יָדַֿע הוָא מֶנ אַימֶכָּא הוָא משַמשָנֶא דֶּינ יָדֿעִינ הוַו דּהֶנֻונ מלַו אֶנֻונ למַיָא קרָא רִיש סמָכָֿא לחַתֿנָא
וֶאמַר לֶה כֻּל אנָש לֻוקדַֿמ חַמרָא טָבָֿא מַיתֶּא ומָא דַּרוִיו הָידֶּינ אַינָא דַּבֿצִיר אַנתּ דֶּינ נטַרתָּיהי לחַמרָא טָבָֿא עדַֿמָא להָשָא
הָדֶֿא הי אָתָֿא קַדֿמָיתָּא דַּעבַֿדֿ יֶשֻוע בּקָטנֶא דַּגֿלִילָא וַאודַּע שֻובֿחֶה והַימֶנו בֶּה תַּלמִידַֿוהי
בָּתַֿר הָדֶֿא נחֶתֿ לַכֿפַֿרנַחֻומ הֻו וֶאמֶה וַאחַוהי ותַֿלמִידַֿוהי ותַֿמָנ הוַו קַלִיל יַומָתָֿא
וקַרִיבֿ הוָא פֶּצחָא דִּיהֻודָֿיֶא וַסלֶק לֻאורִשלֶמ יֶשֻוע
וֶאשכַּח בּהַיכּלָא להָלֶינ דַּמזַבּנִינ תַּורֶא ועֶרבֶּא ויַונֶא ולַמעַרפָֿנֶא דּיָתֿבִּינ
וַעבַֿדֿ לֶה פּרָגֶֿלָא מֶנ חַבֿלָא וַלכֻֿלהֻונ אַפֶּק מֶנ הַיכּלָא וַלעֶרבֶּא וַלתַֿורֶא ולַמעַרפָֿנֶא וֶאשַדֿ עֻורפָֿנהֻונ ופָֿתֻֿורַיהֻונ הפַֿכֿ
וַלהָנֻונ דַּמזַבּנִינ יַונֶא אֶמַר שקֻולו הָלֶינ מֶכָּא ולָא תֶּעבּדֻֿונֶה לבַֿיתֶּה דָּאבֿי בֶּיתֿ תֶּאגֻֿורתָּא
וֶאתּדּכַֿרו תַּלמִידַֿוהי דַּכֿתִֿיבֿ דַּטנָנֶה דּבַֿיתָּכֿ אַכֿלַני
ענַו דֶּינ יִהֻודָֿיֶא וֶאמַרו לֶה מָנָא אָתָֿא מחַוֶא אַנתּ לַנ דּהָלֶינ עָבֶֿדֿ אַנתּ
ענָא יֶשֻוע וֶאמַר להֻונ סתֻֿורו הַיכּלָא הָנָא ולַתֿלָתָֿא יַומִינ אֶנָא מקִימ אנָא לֶה
אָמרִינ לֶה יִהֻודָֿיֶא לַארבּעִינ ושֶתֿ שנִינ אֶתֿבּנִי הַיכּלָא הָנָא וַאנתּ לַתֿלָתָֿא יַומִינ מקִימ אַנתּ לֶה
הֻו דֶּינ אֶמַר הוָא עַל הַיכּלָא דּפַֿגֿרֶה
כַּדֿ קָמ דֶּינ מֶנ בֶּיתֿ מִיתֶֿא אֶתּדּכַֿרו תַּלמִידַֿוהי דּהָדֶֿא אֶמַר הוָא והַימֶנו לַכֿתָֿבֶֿא וַלמֶלתָֿא דֶּאמַר יֶשֻוע
כַּדֿ אִיתַֿוהי הוָא דֶּינ יֶשֻוע בֻּאורִשלֶמ בּפֶֿצחָא בּעַדֿעִדָֿא סַגִּיֶאא הַימֶנו בֶּה דַּחזַו אָתֿוָתָֿא דּעָבֶֿדֿ
הֻו דֶּינ יֶשֻוע לָא מהַימֶנ הוָא להֻונ נַפֿשֶה מֶטֻל דּהֻו יָדַֿע הוָא לכֻֿלנָש
ולָא סנִיק הוָא דּאנָש נֶסהַדֿ לֶה עַל כֻּל בַּרנָש הֻו גֶּיר יָדַֿע הוָא מָנָא אִיתֿ בּבַֿרנָשָא
