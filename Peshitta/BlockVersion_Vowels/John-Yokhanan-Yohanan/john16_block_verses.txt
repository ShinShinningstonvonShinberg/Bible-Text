נַפּקֻונָכֻֿונ גֶּיר מֶנ כּנֻושָתֿהֻונ ותִֿאתֶֿא שָעתָֿא דּכֻֿל דּנֶקטֻולכֻֿונ נַסבַּר דּקֻורבָּנָא מקַרֶבֿ לַאלָהָא
והָלֶינ נֶעבּדֻֿונ מֶטֻל דּלָא יִדַֿעו ולָא לָאבֿי ולָא לִי
הָלֶינ מַללֶתֿ עַמכֻֿונ דּמָא דֶּאתָֿא עֶדָּנהֶינ תֶּעהדֻּונ אֶנֶינ דֶּאנָא אֶמרֶתֿ לכֻֿונ הָלֶינ דֶּינ מֶנ קדִֿימ לָא אֶמרֶתֿ לכֻֿונ דּעַמכֻֿונ הוִיתֿ
הָשָא דֶּינ אָזֶל אנָא לוָתֿ מַנ דּשַדּרַני ולָא אנָש מֶנכֻֿונ משַאֶל לִי לַאיכָּא אָזֶל אַנתּ
אֶמרֶתֿ לכֻֿונ גֶּיר הָלֶינ וֶאתָֿתֿ כַּריֻותָֿא וַמלָתֿ לֶבַּוָתֿכֻֿונ
אֶלָא אֶנָא שרָרָא אָמַר אנָא לכֻֿונ דּפַֿקָח לכֻֿונ דֶּאנָא אִזַל אֶנ גֶּיר אֶנָא לָא אָזֶל אנָא פַּרַקלִטָא לָא אָתֶֿא לוָתֿכֻֿונ אֶנ דֶּינ אִזַל אֶשַדּרִיוהי לוָתֿכֻֿונ
ומָא דֶּאתָֿא הֻו נַכּסִיוהי לעָלמָא עַל חטִיתָֿא ועַל זַדִּיקֻותָֿא ועַל דִּינָא
עַל חטִיתָֿא דּלָא מהַימנִינ בִּי
עַל זַדִּיקֻותָֿא דֶּינ דַּלוָתֿ אָבֿי אָזֶל אנָא ולָא תֻּובֿ חָזֶינ אנתֻּונ לִי
עַל דִּינָא דֶּינ דַּארכֻֿונָא דּעָלמָא הָנָא דִּינ הֻו
תֻּובֿ סַגִּי אִיתֿ לִי למִאמַר לכֻֿונ אֶלָא לָא מֶשכּחִינ אנתֻּונ למֶאחַדֿ הָשָא
מָא דֶּאתָֿא דֶּינ רֻוחָא דַּשרָרָא הֻו נדַֿבַּרכֻֿונ בּכֻֿלֶה שרָרָא לָא גֶּיר נמַלֶל מֶנ רֶעיָנ נַפֿשֶה אֶלָא כֻּל דּנֶשמַע הַו נמַלֶל וַעתִֿידָֿתָֿא נַודַּעכֻֿונ
והֻו נשַבּחַני מֶטֻל דּמֶנ דִּילי נֶסַבֿ וַנחַוֶיכֻֿונ
כֻּל מֶדֶּמ דִּאיתֿ לָאבֿי דִּילי הֻו מֶטֻל הָנָא אֶמרֶתֿ לכֻֿונ דּמֶנ דִּילי נֶסַבֿ וַנחַוֶיכֻֿונ
קַלִיל ולָא תֶּחזֻונָני ותֻֿובֿ קַלִיל ותֶֿחזֻונָני דָּאזֶל אנָא לוָתֿ אַבָֿא
וֶאמַרו הוַו תַּלמִידַֿוהי חַדֿ לחַדֿ מָנָא הי הָדֶֿא דֶּאמַר לַנ דּקַלִיל ולָא תֶּחזֻונָני ותֻֿובֿ קַלִיל ותֶֿחזֻונָני ודָֿאזֶל אנָא לוָתֿ אָבֿי
וָאמרִינ הוַו מָנָא הי הָדֶֿא קַלִיל דֶּאמַר לָא יָדֿעִינַנ מָנָא ממַלֶל
יֶשֻוע דֶּינ יִדַֿע דּבָֿעֶינ הוַו לַמשַאָלֻותֶֿה וֶאמַר להֻונ עַל הָדֶֿא בָּעֶינ אנתֻּונ עַמ חדָֿדֶֿא דֶּאמרֶתֿ לכֻֿונ דּקַלִיל ולָא תֶּחזֻונָני ותֻֿובֿ קַלִיל ותֶֿחזֻונָני
אַמִינ אַמִינ אָמַר אנָא לכֻֿונ דּתֶֿבֿכֻּונ אַנתֻּונ ותֶֿאלֻונ ועָלמָא נֶחדֶּא וַלכֻֿונ תֶּכֿרֶא אֶלָא כַּריֻותֿכֻֿונ לחַדֻֿותָֿא תֶּהוֶא
אַנתּתָֿא מָא דּיָלדָּא כָּריָא לָה דַּמטָא יַומָא דּמַולָדָֿה מָא דּיֶלדַּתֿ דֶּינ בּרָא לָא עָהדָּא אֻולצָנָה מֶטֻל חַדֻֿותָֿא דֶּאתִֿילֶדֿ בַּרנָשָא בּעָלמָא
אָפֿ אַנתֻּונ הָשָא כַּריָא לכֻֿונ תֻּובֿ דֶּינ אֶחזֶיכֻֿונ ונֶחדֶּא לֶבּכֻֿונ וחַדֿוַתֿכֻֿונ לָא אנָש נָסֶבֿ מֶנכֻֿונ
וַבֿהַו יַומָא לִי לָא תֶּשֶאלֻונ מֶדֶּמ אַמִינ אַמִינ אָמַר אנָא לכֻֿונ דּכֻֿל מֶדֶּמ דּתֶֿשֶאלֻונ לָאבֿי בּשֶמי נֶתֶּל לכֻֿונ
עדַֿמָא להָשָא לָא שֶאלתֻּונ מֶדֶּמ בּשֶמי שַאלו ותֶֿסבֻֿונ דּתֶֿהוֶא חַדֿוַתֿכֻֿונ משַמליָא
הָלֶינ בּפֶֿלָאתָֿא מַללֶתֿ עַמכֻֿונ אָתֿיָא דֶּינ שָעתָֿא אֶמַתֿי דּלָא אֶמַלֶל עַמכֻֿונ בּפֶֿלָאתָֿא אֶלָא עִינ בַּגֿלֶא אֶבַֿדֶּק לכֻֿונ עַל אַבָֿא
בּהַו יַומָא דּתֶֿשֶאלֻונ בּשֶמי ולָא אָמַר אנָא לכֻֿונ דֶּאנָא אֶבֿעֶא מֶנ אַבָֿא עלַיכֻּונ
הֻו גֶּיר אַבָֿא רָחֶמ לכֻֿונ דַּאנתֻּונ רחֶמתֻּונָני והַימֶנתֻּונ דֶּאנָא מֶנ לוָתֿ אַלָהָא נֶפֿקֶתֿ
נֶפֿקֶתֿ מֶנ לוָתֿ אַבָֿא וֶאתִֿיתֿ לעָלמָא ותֻֿובֿ שָבֶֿק אנָא לעָלמָא וָאזֶל אנָא לִי לוָתֿ אַבָֿא
אָמרִינ לֶה תַּלמִידַֿוהי הָא הָשָא גַּליָאיִתֿ ממַלֶל אַנתּ ופֶֿלֶאתָֿא ולָא חדָֿא אָמַר אַנתּ
