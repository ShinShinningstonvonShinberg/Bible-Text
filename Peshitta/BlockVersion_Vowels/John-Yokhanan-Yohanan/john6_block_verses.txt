אֶלָא אִיתֿ אנָשָא מֶנכֻֿונ דּלָא מהַימנִינ יָדַֿע הוָא גֶּיר יֶשֻוע מֶנ קדִֿימ מַנ אֶנֻונ אַילֶינ דּלָא מהַימנִינ ומַנֻו הַו דּמַשלֶמ לֶה
וֶאמַר הוָא להֻונ מֶטֻל הָנָא אֶמרֶתֿ לכֻֿונ דּלָא אנָש מֶשכַּח דּנִאתֶֿא לוָתֿי אֶלָא יִהִיבֿ לֶה מֶנ אָבֿי
מֶטֻל הָדֶֿא מֶלתָֿא סַגִּיֶאא מֶנ תַּלמִידַֿוהי אֶזַלו לבֶֿסתַּרהֻונ ולָא מהַלכִֿינ הוַו עַמֶה
וֶאמַר יֶשֻוע לַתֿרֶעסַרתֶּה למָא אָפֿ אַנתֻּונ צָבֶֿינ אנתֻּונ למִאזַל
ענָא שֶמעֻונ כִּאפָֿא וֶאמַר מָרי לוָתֿ מַנ נִאזַל מֶלֶא דּחַיֶא דַּלעָלַמ אִיתֿ לָכֿ
וַחנַנ הַימֶננ וִידַֿענ דַּאנתּ הֻו משִיחָא בּרֶה דַּאלָהָא חַיָא
אָמַר להֻונ יֶשֻוע לָא הוָא אֶנָא גּבִֿיתֿכֻֿונ לַתֿרֶעסַר ומֶנכֻֿונ חַדֿ סָטָנָא הֻו
אֶמַר הוָא דֶּינ עַל יִהֻודָֿא בַּר שֶמעֻונ סכַֿריֻוטָא הֻו גֶּיר עתִֿידֿ הוָא דּנַשלמִיוהי חַדֿ מֶנ תּרֶעסַר
