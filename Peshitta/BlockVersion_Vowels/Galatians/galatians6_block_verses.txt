אַחַי אֶנ אנָש מֶנכֻֿונ נֶתֿקַדַּמ בּסַכֿלֻותָֿא אַנתֻּונ אַילֶינ דַּבֿרֻוח אַנתֻּונ אַתֿקנֻוהי בּרֻוחָא מַכִּיכֿתָּא וַהוַיתֻּונ זהִירִינ דּדַֿלמָא אָפֿ אַנתֻּונ תֶּתֿנַסֻונ
וַטעַנו יֻוקרָא דַּחדָֿדֶֿא דּהָכַֿנָא ממַלֶיתֻּונ נָמֻוסֶה דַּמשִיחָא
אֶנ גֶּיר אנָש סָבַֿר דִּאיתַֿוהי מֶדֶּמ כַּדֿ לָא אִיתַֿוהי נַפֿשֶה מַטעֶא
אֶלָא אנָש עבָֿדֶֿה נֶהוֶא בָּקֶא והָידֶּינ בַּינַוהי לנַפֿשֶה נֶהוֶא שֻובֿהָרֶה ולָא בַּאחרָנֶא
כֻּלנָש גֶּיר מַובּלָא דּנַפֿשֶה נֶשקֻול
נֶשתַּותַּפֿ דֶּינ הַו דּשָמַע מֶלתָֿא להַו מַנ דּמַשמַע לֶה בּכֻֿלהֶינ טָבָֿתָֿא
לָא תֶּטעֻונ אַלָהָא לָא מֶתֿבַּזַח מֶדֶּמ דּזָרַע גֶּיר בַּר אנָשָא הַו הֻו חָצֶדֿ
מַנ דּבַֿבֿסַר זָרַע מֶנ בֶּסרָא חבָֿלָא הו חָצֶדֿ ומַנ דַּבֿרֻוח זָרַע מֶנ רֻוחָא חַיֶא דַּלעָלַמ נֶחצֻודֿ
וכַֿדֿ עָבֿדִּינַנ דּטָבֿ לָא הוָתֿ מָאנָא לַנ נֶהוֶא גֶּיר זַבֿנָא דּנֶחצֻודֿ ולָא תֶּמַאנ לַנ
הָשָא הָכִֿיל עַדֿ זַבֿנָא אִיתֿ לַנ נֶפֿלֻוח טָבָֿתָֿא לוָתֿ כֻּלנָש יַתִּירָאיִתֿ לוָתֿ בּנַי בַּיתָּא דּהַימָנֻותָֿא
חזַו אַילֶינ כּתִֿיבָֿנ כֶּתֿבֶּתֿ לכֻֿונ בִּאידַֿי
אַילֶינ דּצָבֶֿינ דּנֶשתַּבֿהרֻונ בּבֶֿסרָא הֶנֻונ אָלצִינ לכֻֿונ דּתֶֿתֿגַּזרֻונ דּבַֿלחֻודֿ בַּזקִיפֶֿה דַּמשִיחָא לָא נֶתֿרַדֿפֻּונ
אָפֿלָא גֶּיר הֶנֻונ הָלֶינ דּגָֿזרִינ נָטרִינ נָמֻוסָא אֶלָא צָבֶֿינ דּתֶֿתֿגַּזרֻונ דַּבֿבֶֿסרכֻֿונ דִּילכֻֿונ נֶשתַּבֿהרֻונ
לִי דֶּינ לָא נֶהוֶא לִי דֶּאשתַּבֿהַר אֶלָא בַּזקִיפֶֿה דּמָרַנ יֶשֻוע משִיחָא דּבֶֿה עָלמָא זקִיפֿ לִי וֶאנָא זקִיפֿ אנָא לעָלמָא
לָא גֶּיר גּזֻורתָּא אִיתֶֿיה מֶדֶּמ ולָא עֻורלֻותָֿא אֶלָא בּרִיתָֿא חדַֿתָֿא
וַאילֶינ דַּלהָנָא שבִֿילָא שָלמִינ שלָמָא נֶהוֶא עלַיהֻונ ורַחמֶא ועַל אִיסרָיֶל דַּאלָהָא
מֶכִּיל לִי אנָש עַמלָא לָא נַרמֶא אֶנָא גֶּיר כֻּותֿמָתֶֿה דּמָרַנ יֶשֻוע בּפַֿגֿרי שקִיל אנָא
טַיבֻּותֶֿה דּמָרַנ יֶשֻוע משִיחָא עַמ רֻוחכֻֿונ אַחַי אַמִינ
