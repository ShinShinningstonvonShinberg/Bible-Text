וַבֿהֻונ בּיַומָתָֿא הָנֻונ כַּדֿ סגִֿיו תַּלמִידֶֿא רַטֶנו הוַו יַונָיֶא תַּלמִידֶֿא עַל עֶבֿרָיֶא דּמֶתֿבַּסיָנ הוַי אַרמלָתֿהֻונ בּתֶֿשמֶשתָּא דּכֻֿליֻומ
וַקרַו תּרֶעסַר שלִיחֶא לכֻֿלֶה כֶּנשָא דּתַֿלמִידֶֿא וֶאמַרו להֻונ לָא שַפִּיר דּנֶשבֻּוק מֶלתָֿא דַּאלָהָא וַנשַמֶש פָּתֻֿורֶא
בּצַו הָכִֿיל אַחַי וַגֿבַֿו שַבֿעָא גַּבֿרִינ מֶנכֻֿונ דִּאיתֿ עלַיהֻונ סָהדֻּותָֿא וַמלֶינ רֻוחֶה דּמָריָא וחֶכֿמתָֿא וַנקִימ אֶנֻונ עַל הָדֶֿא צבֻֿותָֿא
וַחנַנ נֶהוֶא אַמִינִינ בַּצלֻותָֿא וַבֿתֶֿשמֶשתָּא דּמֶלתָֿא
ושֶפֿרַתֿ הָדֶֿא מֶלתָֿא קדָֿמ כֻּלֶה עַמָא וַגֿבַֿו לֶאסטֶפַֿנָוס גַּבֿרָא דַּמלֶא הוָא הַימָנֻותָֿא ורֻוחָא דּקֻודֿשָא וַלפִֿילִיפָּוס ולַפּרָכָֿרָוס וַלנִיקַנָור וַלטִימָונ וַלפַּרמֶנַא וַלנִיקָלַאוָס גִּיֻורָא אַנטִיָוכָֿיָא
הָלֶינ קָמו קדָֿמַיהֻונ דַּשלִיחֶא וכַֿדֿ צַלִיו סָמו עלַיהֻונ אִידָֿא
ומֶלתֶֿה דַּאלָהָא רָבֿיָא הוָתֿ וסָגֶֿא הוָא מֶניָנָא דּתַֿלמִידֶֿא בֻּאורִשלֶמ טָבֿ ועַמָא סַגִּיָאא מֶנ יִהֻודָֿיֶא מֶשתּמַע הוָא להַימָנֻותָֿא
אֶסטֶפַֿנָוס דֶּינ מלֶא הוָא טַיבֻּותָֿא וחַילָא ועָבֶֿדֿ הוָא אָתֿוָתָֿא ותֶֿדֿמרָתָֿא בּעַמָא
וקָמו הוַו אנָשָא מֶנ כּנֻושתָּא דּמֶתֿקַריָא דּלִיבֶֿרטִינֻו וקֻורִינָיֶא וַאלֶכּסַנדּרָיֶא וַדֿמֶנ קִילִיקִיַא ומֶנ אַסִיַא ודָֿרשִינ הוַו עַמ אֶסטֶפַֿנָוס
ולָא מֶשכּחִינ הוַו לַמקָמ לֻוקבַֿל חֶכֿמתָֿא ורֻוחָא דַּממַללָא הוָתֿ בֶּה
הָידֶּינ שַדַּרו לגַֿבֿרֶא וַאלֶפֿו אֶנֻונ דּנִאמרֻונ דַּחנַנ שמַענָיהי דֶּאמַר מֶלֶא דּגֻֿודָּפָֿא עַל מֻושֶא ועַל אַלָהָא
וַשגַֿשו לעַמָא וַלקַשִישֶא וַלסָפֿרֶא וֶאתַֿו וקָמו עלַוהי וַחטַפֿו אַיתּיֻוהי למֶצעַתֿ כֶּנשָא
וַאקִימו סָהדֶּא דַּגָּלֶא דָּאמרִינ הָנָא גַּבֿרָא לָא שָלֶא לַממַלָלֻו מֶלֶא לֻוקבַֿל נָמֻוסָא ועַל אַתֿרָא הָנָא קַדִּישָא
חנַנ גֶּיר שמַענָיהי דֶּאמַר דּיֶשֻוע הָנָא נָצרָיָא הֻו נֶשרֶיוהי לַאתֿרָא הָנָא וַנחַלֶפֿ עיָדֶֿא דַּאשלֶמ לכֻֿונ מֻושֶא
וחָרו בֶּה כֻּלהֻונ הָנֻונ דּיָתֿבִּינ הוַו בַּכֿנֻושתָּא וַחזַו פַּרצֻופֶּה אַיכֿ פַּרצֻופָּא דּמַלַאכָֿא
