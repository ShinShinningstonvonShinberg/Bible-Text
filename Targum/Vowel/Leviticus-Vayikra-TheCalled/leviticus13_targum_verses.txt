וִיהֵי מַכְתָּשָׁא יָרוֹק אוֹ סַמּוֹק בִּלְבוּשָׁא אוֹ בְמַשְׁכָּא אוֹ בְשִׁתְיָא אוֹ בְעַרְבָא אוֹ בְכָל מַאן דִּמְשַׁךְ מַכְתַּשׁ סְגִירוּתָא הוּא וְיִתַּחֲזֵי לְכַהֲנָא
וְיֶחֱזֵי כַהֲנָא יָת מַכְתָּשָׁא וְיַסְגַּר יָת מַכְתָּשָׁא שִׁבְעַת יוֹמִין
וְיֶחֱזֵי יָת מַכְתָּשָׁא בְּיוֹמָא שְׁבִיעָאָה אֲרֵי אוֹסֵף מַכְתָּשָׁא בִּלְבוּשָׁא אוֹ בְשִׁתְיָא אוֹ בְעַרְבָא אוֹ בְמַשְׁכָּא לְכֹל דִּי יִתְעֲבֵד מַשְׁכָּא לְעִבִידָא סְגִירוּת מְחַסְרָא מַכְתָּשָׁא מְסָאָב הוּא
וְיוֹקִיד יָת לְבוּשָׁא אוֹ יָת שִׁתְיָא אוֹ יָת עַרְבָא בְּעַמְרָא אוֹ בְכִתָּנָא אוֹ יָת כָּל מָאן דִּמְשַׁךְ דִּי יְהֵי בֵיהּ מַכְתָּשָׁא אֲרֵי סְגִירוּת מְחַסְרָא הִיא בְּנוּרָא תִּתּוֹקָד
וְאִם יֶחֱזֵי כַהֲנָא וְהָא לָא אוֹסֵף מַכְתָּשָׁא בִּלְבוּשָׁא אוֹ בְשִׁתְיָא אוֹ בְעַרְבָא אוֹ בְּכָל מָאן דִּמְשָׁךְ
וִיפַקֵּד כַּהֲנָא וִיחַוְּרוּן יָת דִּי בֵיהּ מַכְתָּשָׁא וְיַסְגְּרִנֵּיהּ שִׁבְעַת יוֹמִין תִּנְיָנוּת
וְיֶחֱזֵי כַהֲנָא בָּתַר דְּחַוָּרוּ יָת מַכְתָּשָׁא וְהָא לָא שְׁנָא מַכְתָּשָׁא מִן כַּד הֲוָה וּמַכְתָּשָׁא לָא אוֹסֵף מְסָאָב הוּא בְּנוּרָא תּוֹקְדִנֵּיהּ תַּבְרָא הִיא בְּשְׁחִיקוּתֵיהּ אוֹ בְחַדָּתוּתֵיהּ
וְאִם חֲזָא כַהֲנָא וְהָא עָמְיָא מַכְתָּשָׁא בָּתַר דְּחַוָּרוּ יָתֵיהּ וּבְזַע יָתֵיהּ מִן לְבוּשָׁא אוֹ מִן מַשְׁכָּא אוֹ מִן שִׁתְיָא אוֹ מִן עַרְבָא
וְאִם תִּתְחֲזֵי עוֹד בִּלְבוּשָׁא אוֹ בְשִׁתְיָא אוֹ בְעַרְבָא אוֹ בְכָל מָאן דִּמְשַׁךְ סָגְיָא הִיא בְּנוּרָא תוֹקְדִנֵּיהּ יָת דִּי בֵיהּ מַכְתָּשָׁא
וּלְבוּשָׁא אוֹ שִׁתְיָא אוֹ עַרְבָא אוֹ כָל מָאן דִּמְשַׁךְ דִּי יִתְחַוַּר וְיֶעְדֵּי מִנְּהוֹן מַכְתָּשָׁא וְיִצְטַבַּע תִּנְיָנוּת וְיִדְכֵּי
דָּא אוֹרַיְתָא דְמַכְתַּשׁ סְגִירוּ בִּלְבוּשׁ עֲמַר אוֹ כִתָּנָא אוֹ שִׁתְיָא אוֹ עַרְבָא אוֹ כָּל מָאן דִּמְשָׁךְ לְדַכָּיוּתֵיהּ אוֹ לְסַאֲבוּתֵיהּ
